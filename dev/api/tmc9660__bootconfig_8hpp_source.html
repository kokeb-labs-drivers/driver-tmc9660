175class BootConfig {
176public:
178 static constexpr std::size_t kBytes = 64;
180 static constexpr std::size_t kWords = 16;
181
183 constexpr BootConfig() noexcept = default;
184
189 explicit constexpr BootConfig(const std::array<uint32_t, kWords>& words) noexcept
190 : words_(words) {}
191
199 [[nodiscard]] constexpr Error Set(BootField field, uint32_t value) noexcept {
200 if (!detail::IsEnumerator(field)) {
201 return Error::InvalidArgument;
202 }
203 const detail::BootFieldInfo info = fieldInfo(field);
204 if (value > maxValue(info)) {
205 return Error::InvalidArgument;
206 }
207 const uint32_t shift = bitShift(info);
208 const uint32_t mask = maxValue(info) << shift;
209 uint32_t& word = words_[wordIndex(info)];
210 word = (word & ~mask) | ((value << shift) & mask);
211 return Error::Ok;
212 }
213
219 [[nodiscard]] constexpr uint32_t Get(BootField field) const noexcept {
220 if (!detail::IsEnumerator(field)) {
221 return 0U;
222 }
223 const detail::BootFieldInfo info = fieldInfo(field);
224 return (words_[wordIndex(info)] >> bitShift(info)) & maxValue(info);
225 }
226
232 [[nodiscard]] static constexpr uint32_t MaxOf(BootField field) noexcept {
233 return detail::IsEnumerator(field) ? maxValue(fieldInfo(field)) : 0U;
234 }
235
241 [[nodiscard]] constexpr uint32_t Word(std::size_t index) const noexcept {
242 return (index < kWords) ? words_[index] : 0U;
243 }
244
251 [[nodiscard]] constexpr Error SetWord(std::size_t index, uint32_t value) noexcept {
252 if (index >= kWords) {
253 return Error::InvalidArgument;
254 }
255 words_[index] = value;
256 return Error::Ok;
257 }
258
260 [[nodiscard]] constexpr const std::array<uint32_t, kWords>& Words() const noexcept {
261 return words_;
262 }
263
270 [[nodiscard]] constexpr Error CheckStart() const noexcept {
271 const uint32_t mode = Get(BootField::BootMode);
272 if (Get(BootField::StartMotorCtrl) == 0U) {
273 return Error::Ok;
274 }
275 if ((mode != static_cast<uint32_t>(BootMode::Register)) &&
276 (mode != static_cast<uint32_t>(BootMode::Parameter))) {
277 return Error::Refused;
278 }
279 if (Get(BootField::ClockPllOutSel) != 1U) {
280 return Error::Refused;
281 }
282 if (Get(BootField::ClockSysClkDiv) != 0U) {
283 return Error::Refused;
284 }
285 return Error::Ok;
286 }
287
293 [[nodiscard]] constexpr bool operator==(const BootConfig& other) const noexcept {
294 return words_ == other.words_;
295 }
296
297private:
298 [[nodiscard]] static constexpr detail::BootFieldInfo fieldInfo(BootField f) noexcept {
299 return detail::kBootFields[static_cast<std::size_t>(detail::ToUnderlying(f))];
300 }
301 [[nodiscard]] static constexpr uint32_t maxValue(const detail::BootFieldInfo& i) noexcept {
302 return (uint32_t{1} << i.width) - 1U;
303 }
304 [[nodiscard]] static constexpr std::size_t wordIndex(const detail::BootFieldInfo& i) noexcept {
305 return static_cast<std::size_t>(i.offset / 4U);
306 }
307 [[nodiscard]] static constexpr uint32_t bitShift(const detail::BootFieldInfo& i) noexcept {
308 return (static_cast<uint32_t>(i.offset % 4U) * 8U) + i.shift;
309 }

311 std::array<uint32_t, kWords> words_{};
312};

341[[nodiscard]] constexpr bool PartitionsValid(std::span<const PartitionEntry> entries,
342 uint64_t sector) noexcept {
343 for (std::size_t i = 0U; i < entries.size(); ++i) {
344 const PartitionEntry& e = entries[i];
345 if (((e.type != 2U) && (e.type != 4U)) || (e.size == 0U) || ((e.offset % sector) != 0U) ||
346 ((e.size % sector) != 0U) || ((e.type == 2U) && (e.size < 0x400U))) {
347 return false;
348 }
349 for (std::size_t j = 0U; j < i; ++j) {
350 const PartitionEntry& o = entries[j];
351 const uint64_t a0 = e.offset;
352 const uint64_t a1 = a0 + e.size;
353 const uint64_t b0 = o.offset;
354 const uint64_t b1 = b0 + o.size;
355 if ((a0 < b1) && (b0 < a1)) {
356 return false;
357 }
358 }
359 }
360 return true;
361}

364constexpr void PutLe32(std::span<uint8_t> out, std::size_t at, uint32_t v) noexcept {
365 out[at] = static_cast<uint8_t>(v);
366 out[at + 1U] = static_cast<uint8_t>(v >> 8U);
367 out[at + 2U] = static_cast<uint8_t>(v >> 16U);
368 out[at + 3U] = static_cast<uint8_t>(v >> 24U);
369}
370
372constexpr void WritePartitionEntries(std::span<const PartitionEntry> entries,
373 std::span<uint8_t> out) noexcept {
374 for (std::size_t i = 0U; i < entries.size(); ++i) {
375 const std::size_t at = kPartitionHeaderBytes + (kPartitionEntryBytes * i);
376 for (std::size_t k = 0U; k < 12U; ++k) {
377 out[at + k] = static_cast<uint8_t>(entries[i].name[k]);
378 }
379 out[at + 12U] = static_cast<uint8_t>(entries[i].type | (entries[i].writable ? 0x80U : 0U));
380 PutLe32(out, at + 16U, entries[i].offset);
381 PutLe32(out, at + 20U, entries[i].size);
382 }
383}

398[[nodiscard]] constexpr Result<std::size_t> BuildPartitionTable(
399 std::span<const PartitionEntry> entries, uint8_t size_index, uint8_t sector_index,
400 std::span<uint8_t> out) noexcept {
401 const std::size_t total = kPartitionHeaderBytes + (kPartitionEntryBytes * entries.size());
402 if (entries.empty() || (out.size() < total) || (size_index > 31U) || (sector_index > 31U)) {
403 return Result<std::size_t>::Failure(Error::InvalidArgument);
404 }
405 const uint64_t sector = (sector_index == 0U) ? 1ULL : (1ULL << sector_index);
406 if (!detail::PartitionsValid(entries, sector)) {
407 return Result<std::size_t>::Failure(Error::InvalidArgument);
408 }
409 for (std::size_t i = 0U; i < total; ++i) {
410 out[i] = 0U;
411 }
412 detail::PutLe32(out, 0U, static_cast<uint32_t>(total));
413 out[8] = static_cast<uint8_t>(0xFE);
414 out[9] = static_cast<uint8_t>(0xCA);
415 out[10] = static_cast<uint8_t>(1);
416 out[11] = static_cast<uint8_t>(1);
417 out[12] = size_index;
418 out[13] = sector_index;
419 detail::WritePartitionEntries(entries, out);
420 detail::PutLe32(out, 4U, detail::Crc32(std::span<const uint8_t>{out}.subspan(8U, total - 8U)));
421 return Result<std::size_t>::Success(total);
422}
