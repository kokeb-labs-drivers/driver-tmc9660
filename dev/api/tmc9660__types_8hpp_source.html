495[[nodiscard]] constexpr Error Validate(const Config& c) noexcept {
496 if ((c.device_address == 0U) || (c.host_address == 0U) || (c.tmcl_host_address == 0U)) {
497 return Error::InvalidArgument;
498 }
499 if ((c.reply_timeout_us == 0U) || (c.boot_timeout_ms == 0U) || (c.spi_reply_retries > 50U)) {
500 return Error::InvalidArgument;
501 }
502 if ((c.canary_variable > 15U) || (c.quirks.ram_debug_index_bits < 5U) ||
503 (c.quirks.ram_debug_index_bits > 6U)) {
504 return Error::InvalidArgument;
505 }
506 if (!detail::IsEnumerator(c.mode) || !detail::IsEnumerator(c.quirks.tmcl_layout)) {
507 return Error::InvalidArgument;
508 }
509 return Error::Ok;
510}
