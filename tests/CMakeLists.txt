# =============================================================================
# TMC9660 driver: host tests
# =============================================================================
# The driver against a register model of the chip, with the firmware's flags
# (-fno-exceptions -fno-rtti) and warnings as errors. No hardware, no ESP-IDF:
#
#   cmake -S . -B build -DTMC9660_BUILD_TESTS=ON
#   cmake --build build
#   ctest --test-dir build --output-on-failure
# =============================================================================

# ISO C++20, never the GNU dialect (handbook KB-CPP20-001): set here too, so that this folder is strict
# whoever includes it.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# -pedantic-errors and the C++20 deprecations as errors (KB-CPP20-001, -003). -Wswitch-enum: a switch over
# an enumeration ends with `default: break;` (MISRA C++:2023 Rule 9.4.2), and a new enumerator still warns.
set(TMC9660_TEST_FLAGS -fno-exceptions -fno-rtti -Wall -Wextra -Wpedantic -pedantic-errors -Wshadow
    -Wconversion -Wsign-conversion -Wdeprecated -Wdeprecated-enum-enum-conversion
    -Wdeprecated-enum-float-conversion -Wimplicit-fallthrough -Wswitch-enum -Werror)
# The standard library's own bounds checks, so an out-of-range span or array access traps in the tests
# and fuzzers (KB-CPP20-043).
set(TMC9660_TEST_DEFINITIONS _GLIBCXX_ASSERTIONS=1 _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST)

function(tmc9660_test_target name)
  target_link_libraries(${name} PRIVATE tmc9660::tmc9660)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_compile_options(${name} PRIVATE ${TMC9660_TEST_FLAGS})
  target_compile_definitions(${name} PRIVATE ${TMC9660_TEST_DEFINITIONS})
endfunction()

# --- Strict-dialect header checks ---------------------------------------------
# One translation unit per public header (and through the driver header its src/ .ipp): each header
# compiles on its own, first in its unit, as ISO C++20 with -pedantic-errors, and the unit asserts
# that the dialect really is C++20 (handbook KB-CPP20-001, checks strict-dialect and
# header-self-contained). They build with the host tests on GCC and Clang and, as part of
# tmc9660_optimised_build, in the Cortex-M cross-build.
file(GLOB_RECURSE TMC9660_PUBLIC_HEADERS CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/inc/*.hpp
  ${PROJECT_SOURCE_DIR}/inc/*.h)
set(TMC9660_HEADER_CHECKS)
foreach(header ${TMC9660_PUBLIC_HEADERS} ${TMC9660_VERSION_HEADER})
  if(header STREQUAL TMC9660_VERSION_HEADER)
    file(RELATIVE_PATH rel ${TMC9660_VERSION_HEADER_DIR} ${header})
  else()
    file(RELATIVE_PATH rel ${PROJECT_SOURCE_DIR}/inc ${header})
  endif()
  string(MAKE_C_IDENTIFIER "${rel}" id)
  set(check ${CMAKE_CURRENT_BINARY_DIR}/header_checks/${id}.cpp)
  string(CONCAT content "// Generated by tests/CMakeLists.txt: ${rel} compiles on its own as ISO C++20.\n"
    "#include \"${rel}\"\n"
    "static_assert(__cplusplus == 202002L, \"the strict-dialect check must build as -std=c++20\");\n")
  if(EXISTS ${check})
    file(READ ${check} old)
  else()
    set(old "")
  endif()
  if(NOT old STREQUAL content)
    file(WRITE ${check} "${content}")
  endif()
  list(APPEND TMC9660_HEADER_CHECKS ${check})
endforeach()
add_library(tmc9660_header_checks OBJECT ${TMC9660_HEADER_CHECKS})
tmc9660_test_target(tmc9660_header_checks)

# --- Version: the getters, and every copy of the version string ---------------
function(tmc9660_version_test name include_dir major minor patch)
  add_executable(${name} test_version.cpp)
  if(include_dir)
    target_include_directories(${name} BEFORE PRIVATE ${include_dir})
  endif()
  tmc9660_test_target(${name})
  target_compile_definitions(${name} PRIVATE EXPECTED_MAJOR=${major} EXPECTED_MINOR=${minor}
    EXPECTED_PATCH=${patch} EXPECTED_STRING="${major}.${minor}.${patch}")
  add_test(NAME ${name} COMMAND ${name})
endfunction()

tmc9660_version_test(tmc9660_version "" ${PROJECT_VERSION_MAJOR} ${PROJECT_VERSION_MINOR}
  ${PROJECT_VERSION_PATCH})

# The same template with another version, found ahead of the real header.
function(tmc9660_version_probe_header)
  set(TMC9660_VERSION_MAJOR 9)
  set(TMC9660_VERSION_MINOR 8)
  set(TMC9660_VERSION_PATCH 7)
  set(TMC9660_VERSION 9.8.7)
  configure_file(${TMC9660_VERSION_TEMPLATE} ${CMAKE_CURRENT_BINARY_DIR}/version_probe/tmc9660_version.h @ONLY)
endfunction()
tmc9660_version_probe_header()
tmc9660_version_test(tmc9660_version_probe ${CMAKE_CURRENT_BINARY_DIR}/version_probe 9 8 7)

  list(GET parts 0 file)

  string(MAKE_C_IDENTIFIER "${file}" id)
  add_test(NAME tmc9660_version_copy_${id}

            -P ${CMAKE_CURRENT_SOURCE_DIR}/check_version_copy.cmake)
endforeach()

# --- Compile-time contract: these must fail to build, for the reason in their EXPECT: line ---
file(GLOB TMC9660_COMPILE_FAIL CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/compile_fail/*.cpp)
foreach(src ${TMC9660_COMPILE_FAIL})
  get_filename_component(stem ${src} NAME_WE)
  file(STRINGS ${src} expect REGEX "EXPECT: ")
  string(REGEX REPLACE ".*EXPECT: " "" expect "${expect}")
  add_executable(tmc9660_compile_fail_${stem} EXCLUDE_FROM_ALL ${src})
  tmc9660_test_target(tmc9660_compile_fail_${stem})
  add_test(NAME tmc9660_compile_fail_${stem}
    COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target tmc9660_compile_fail_${stem})
  # Pass only when the compiler rejects the code with the expected diagnostic, not another error.
  set_tests_properties(tmc9660_compile_fail_${stem} PROPERTIES PASS_REGULAR_EXPRESSION "${expect}")
endforeach()

# --- Generated tables: inc/tmc9660_params.hpp and inc/tmc9660_registers.hpp must equal the generator's output ---
# The generator (tools/gen_tables/) formats with the gate's clang-format (version 20). Without one, or with another
# version (the CI runners have 18), it exits 77 and CTest reports the test as skipped, not failed; where clang-format
# 20 is available it is a real check.
find_program(TMC9660_CLANG_FORMAT NAMES clang-format-20 clang-format)
find_package(Python3 COMPONENTS Interpreter)
if(TMC9660_CLANG_FORMAT AND Python3_FOUND)
  add_test(NAME tmc9660_generated_tables
    COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tools/gen_tables/gen_tables.py --check)
  set_tests_properties(tmc9660_generated_tables PROPERTIES
    ENVIRONMENT "CLANG_FORMAT=${TMC9660_CLANG_FORMAT}"
    SKIP_RETURN_CODE 77)
endif()
