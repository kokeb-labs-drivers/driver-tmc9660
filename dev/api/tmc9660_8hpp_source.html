41// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept the exact
42// signature, `noexcept` and the result type; the driver rejects a hook that is declared but does
43// not match (KB-CPP20-007).
44
46template <typename Port>
47concept DeclaresGpioSet =
48 requires(Port& port) { port.GpioSet(CtrlPin::Reset, GpioSignal::ACTIVE); };
49
51template <typename Port>
52concept HasGpioSet =
53 requires { ExactHook<Port, bool(CtrlPin, GpioSignal) noexcept>::Take(&Port::GpioSet); };
54
56template <typename Port>
57concept DeclaresGpioGet =
58 requires(Port& port, bool& level) { port.GpioGet(CtrlPin::Faultn, level); };
59
61template <typename Port>
62concept HasGpioGet =
63 requires { ExactHook<Port, bool(CtrlPin, bool&) noexcept>::Take(&Port::GpioGet); };
64
66template <typename Port>
67concept DeclaresDelayUs = requires(Port& port) { port.DelayUs(static_cast<uint32_t>(1)); };
68
71template <typename Port>
72concept HasDelayUs = requires { ExactHook<Port, void(uint32_t) noexcept>::Take(&Port::DelayUs); };
73
75template <typename Port>
76concept DeclaresLog = requires(Port& port) { port.Log(LogLevel::Info, ""); };
77
79template <typename Port>
80concept HasLog =
81 requires { ExactHook<Port, void(LogLevel, const char*) noexcept>::Take(&Port::Log); };

85template <typename Port>
86void DelayUs(Port& port, uint32_t us) noexcept {
87 if constexpr (HasDelayUs<Port>) {
88 port.DelayUs(us);
89 } else {
90 static_assert(kAlwaysFalse<Port>,
91 "tmc9660: the port must implement `void DelayUs(uint32_t us) noexcept`");
92 }
93}

96template <typename Port>
97void Log(Port& port, LogLevel level, const char* message) noexcept {
98 if constexpr (HasLog<Port>) {
99 port.Log(level, message);
100 } else {
101 // No Log() hook: every parameter is used in this branch too (MISRA C++:2023 Rule 0.2.2).
102 static_cast<void>(port);
103 static_cast<void>(level);
104 static_cast<void>(message);
105 }
